%
\item (Integration by Parts)
For differentiable functions $u\brak{x}$ and $v\brak{x}$, the integration by parts formula is:
\begin{align}
\int u \, dv = u v - \int v \, du 
\label{eq:integration_by_parts}
\end{align}
%
\item Let
\begin{align}
	\vec{V}&=\myvec{u \\ v}, \nabla = \myvec{\frac{\partial }{\partial x} \\[1ex] \frac{\partial }{\partial y}},
	\\
\end{align}
The curl and divergence of $\vec{V}$ are then defined respectively as
\begin{align}
	\label{eq:curl}
    \nabla\times\vec{V}
    &=\brak{\frac{\partial v}{\partial x}
    -\frac{\partial u}{\partial y}}\vec{e}_3
    \\
    \nabla\cdot\vec{V}
    &=\frac{\partial u}{\partial x}
     +\frac{\partial v}{\partial y}.
	\label{eq:div}
\end{align}
%
%
\item	If $B \subset A, AB = 1$. 
	\label{eq:subset}
\item (Conditional Probability) 
\begin{align}
	\Pr\brak{A \mid B} = \frac{\Pr\brak{A \cap B}}{\Pr\brak{B}}, \quad \Pr\brak{B} > 0. 
	\label{eq:cond-prob}
\end{align}
	
\item (Variance) 
\begin{align}
	\sigma^2 &= \mean{X^2} - \brak{\mean{X}}^2 
	\label{eq:discrete_variance}
\end{align}

\end{enumerate}
